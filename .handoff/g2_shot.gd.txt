extends SceneTree
# godot --path shadowfall --audio-driver Dummy -s shot.gd -- --what=cave|overworld --gloom=N --out=x.png [--cell=c,r]

func _initialize() -> void:
	_go.call_deferred()

func _go() -> void:
	var o := {}
	for a in OS.get_cmdline_user_args():
		var kv := a.trim_prefix("--").split("=", true, 1)
		o[kv[0]] = kv[1] if kv.size() > 1 else ""
	var game = root.get_node("Game")
	var scenes = root.get_node("Scenes")
	game.gloom = float(o.get("gloom", "0")) / 3.0
	if o["what"] == "cave":
		scenes.goto("cave", {"file": "caves/02_the_vine_garden.txt"})
	else:
		scenes.goto("overworld")
	for i in 10:
		await process_frame
	var ow = scenes.screen
	if o.has("cell"):
		var c: PackedStringArray = o["cell"].split(",")
		ow.hero.place(Vector2i(int(c[0]), int(c[1])))
		ow.cam.position = ow.hero.position
		ow.cam.reset_smoothing()
	for m in ow.get("monsters") if ow.get("monsters") else []:
		pass
	for i in 40:
		await process_frame
	root.get_viewport().get_texture().get_image().save_png(o["out"])
	quit()
